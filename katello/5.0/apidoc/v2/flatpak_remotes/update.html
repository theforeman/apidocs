PUT /katello/api/flatpak_remotes/:id
Update a flatpak remote

Examples

PUT /katello/api/flatpak_remotes/228288065
{
  "name": "test",
  "flatpak_remote": {
    "name": "test"
  }
}
200
{
  "name": "test",
  "url": "https://flatpaks.redhat.io/rhel/",
  "description": null,
  "username": null,
  "seeded": true,
  "registry_url": null,
  "id": 228288065,
  "organization_id": 114267492,
  "organization": {
    "name": "Empty Organization",
    "label": "Empty_Organization",
    "id": 114267492
  },
  "last_scan": null,
  "upstream_password_exists": false,
  "last_scan_words": null
}

name
